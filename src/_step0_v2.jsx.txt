const Step0Configurator = ({ lang, global, setGlobal, sessions, setSessions, activeIdx, setActiveIdx, onApply, onSkip }) => {
  const active = sessions[activeIdx] || sessions[0];
  const setSession = (patch) => {
    setSessions(prev => prev.map((s, i) => i === activeIdx ? {...s, ...patch} : s));
  };
  const resetFrom = (field, value) => {
    const cascades = {
      product:    { method:null, submethod:null, media:null, views:[], viewsCount:'', quantity:'', postprod:false, postprodVideo:false },
      method:     { submethod:null, media:null, views:[], viewsCount:'', quantity:'', postprod:false, postprodVideo:false },
      submethod:  { media:null, views:[], viewsCount:'', quantity:'', postprod:false, postprodVideo:false },
      media:      { views:[], viewsCount:'', quantity:'', postprod:false, postprodVideo:false },
    };
    setSession({ [field]: value, ...(cascades[field] || {}) });
  };
  const addSession = () => {
    setSessions(prev => [...prev, makeBlankSession()]);
    setActiveIdx(sessions.length);
  };
  const removeSession = (idx) => {
    if (sessions.length <= 1) return;
    setSessions(prev => prev.filter((_, i) => i !== idx));
    setActiveIdx(Math.max(0, Math.min(activeIdx, sessions.length - 2)));
  };

  const sessionValid = (s) => {
    if (!s.product) return false;
    if (s.product === 'cyclorama') return true;
    if (s.product === 'pap') {
      if (!s.method) return false;
      if (s.method === 'packshot') {
        if (!s.submethod) return false;
        if (!(s.views || []).some(v => v !== 'detail')) return false;
        if (!Number(s.quantity)) return false;
        return true;
      }
      if (s.method === 'onmodel') {
        if (!s.media) return false;
        if (!Number(s.quantity)) return false;
        if (!Number(s.viewsCount)) return false;
        return true;
      }
      return false;
    }
    if (s.product === 'accessoires') {
      if (!s.submethod) return false;
      if (!s.media) return false;
      if (!Number(s.quantity)) return false;
      if (!Number(s.viewsCount)) return false;
      return true;
    }
    if (!s.media) return false;
    if (!Number(s.quantity)) return false;
    if (!Number(s.viewsCount)) return false;
    return true;
  };

  const allValid = sessions.every(sessionValid);
  const recs = allValid ? sessions.map(s => ({ session: s, ...recommendSession(s, global) })) : null;

  const S = active;
  const tabGrid = { display:'grid', gridTemplateColumns:`repeat(${sessions.length}, minmax(0,1fr))`, gap:1, background:'#000', borderTop:'1px solid #000', borderBottom:'1px solid #000' };

  return (
    <div>
      {/* HERO */}
      <div style={{padding:'0 48px'}}>
        <StepIntro n="00" lang={lang}
          t={lang==='fr'?'Configurer votre projet':'Configure your project'}
          s={lang==='fr'
            ?"Quelques questions sur vos produits — nous composons le plateau, la durée et l'équipe adaptés. Vous pourrez ajuster ensuite."
            :"A few questions about your products — we compose the right stage, duration and team. You can tweak after."}/>
      </div>

      {/* SKIP */}
      <div style={{display:'grid',gridTemplateColumns:'1fr auto',gap:1,background:'#000',borderTop:'1px solid #000',borderBottom:'1px solid #000'}}>
        <div style={{background:'#fafafa',padding:'14px 48px',display:'flex',alignItems:'center',gap:14,flexWrap:'wrap'}}>
          <span style={{fontFamily:'var(--font-mono)',fontSize:10,letterSpacing:'0.15em',textTransform:'uppercase',color:'#888'}}>
            {lang==='fr'?'Déjà décidé ?':'Already know?'}
          </span>
          <span style={{fontSize:13,color:'#595959',lineHeight:1.35}}>
            {lang==='fr'?'Passez le configurateur et choisissez manuellement.':'Skip and pick manually.'}
          </span>
        </div>
        <button onClick={onSkip} style={{background:'#fff',border:0,padding:'0 24px',cursor:'pointer',fontFamily:'var(--font-mono)',fontSize:11,letterSpacing:'0.15em',textTransform:'uppercase',color:'#141414',display:'flex',alignItems:'center',gap:8}}>
          {lang==='fr'?'Mode manuel':'Manual mode'} <IconArrowRight width="14" height="14"/>
        </button>
      </div>

      {/* SESSION TABS (only if > 1) */}
      {sessions.length > 1 && (
        <>
          <div style={{padding:'28px 48px 8px',display:'flex',alignItems:'baseline',justifyContent:'space-between',gap:16,flexWrap:'wrap'}}>
            <span className="edo-cell-label" style={{color:'var(--edo-orange)'}}>
              {lang==='fr'?'Sessions produit':'Product sessions'} — {sessions.length}
            </span>
            <button onClick={addSession} style={{background:'#fff',border:'1px solid var(--edo-gray-200)',padding:'8px 14px',cursor:'pointer',fontFamily:'var(--font-mono)',fontSize:10,letterSpacing:'0.15em',textTransform:'uppercase',color:'#141414',display:'flex',alignItems:'center',gap:8,height:32}}>
              + {lang==='fr'?'Ajouter une session':'Add a session'}
            </button>
          </div>
          <div style={tabGrid}>
            {sessions.map((s, i) => {
              const isActive = i === activeIdx;
              const valid = sessionValid(s);
              const p = PRODUCTS.find(x => x.k === s.product);
              return (
                <button key={i} onClick={()=>setActiveIdx(i)} style={{
                  background: isActive ? '#141414' : '#fff',
                  color: isActive ? '#fff' : '#141414',
                  border:0,padding:'12px 14px',textAlign:'left',cursor:'pointer',fontFamily:'inherit',
                  display:'flex',flexDirection:'column',gap:3,minWidth:0,
                }}>
                  <div style={{display:'flex',alignItems:'center',justifyContent:'space-between',gap:8}}>
                    <span style={{fontFamily:'var(--font-mono)',fontSize:10,letterSpacing:'0.15em',color:isActive?'var(--edo-orange)':'#888'}}>
                      {lang==='fr'?'Session':'Session'} {String(i+1).padStart(2,'0')}
                    </span>
                    <span onClick={(e)=>{e.stopPropagation(); removeSession(i);}}
                      style={{fontSize:14,color:isActive?'rgba(255,255,255,0.5)':'#888',cursor:'pointer',padding:'0 4px',lineHeight:1}}
                      title={lang==='fr'?'Retirer':'Remove'}>×</span>
                  </div>
                  <div style={{fontSize:14,fontWeight:400,letterSpacing:'-0.015em'}}>
                    {p ? p[lang] : (lang==='fr'?'À définir':'To define')}
                  </div>
                  <div style={{fontFamily:'var(--font-mono)',fontSize:9,letterSpacing:'0.05em',color:isActive?'rgba(255,255,255,0.55)':'#888'}}>
                    {valid ? `${s.quantity} ${lang==='fr'?'produits':'products'}` : (lang==='fr'?'incomplet':'incomplete')}
                  </div>
                </button>
              );
            })}
          </div>
        </>
      )}

      {/* Q1 — TYPE DE PRODUIT */}
      <div style={{padding:'20px 48px 8px'}}>
        <span className="edo-cell-label" style={{color:'var(--edo-orange)'}}>01 · {lang==='fr'?'Type de produit':'Product type'}</span>
      </div>
      <div style={{display:'grid',gridTemplateColumns:'repeat(auto-fit,minmax(160px,1fr))',gap:1,background:'#000',borderTop:'1px solid #000',borderBottom:'1px solid #000'}}>
        {PRODUCTS.map((p,i)=>(
          <CfgChoice key={p.k} idx={i+1} on={S.product===p.k}
            onClick={()=>resetFrom('product', p.k)}
            label={p[lang]} desc={p.desc[lang]}/>
        ))}
      </div>

      {/* Q2 — PAP Méthode */}
      {S.product === 'pap' && (
        <>
          <div style={{padding:'20px 48px 8px'}}>
            <span className="edo-cell-label" style={{color:'var(--edo-orange)'}}>02 · {lang==='fr'?'Méthode':'Method'}</span>
          </div>
          <div style={{display:'grid',gridTemplateColumns:'1fr 1fr',gap:1,background:'#000',borderTop:'1px solid #000',borderBottom:'1px solid #000'}}>
            {PAP_METHODS.map((m,i)=>(
              <CfgChoice key={m.k} idx={i+1} on={S.method===m.k}
                onClick={()=>resetFrom('method', m.k)}
                label={m[lang]} desc={m.desc[lang]}/>
            ))}
          </div>
        </>
      )}

      {/* Q2b — PAP Packshot sous-type */}
      {S.product === 'pap' && S.method === 'packshot' && (
        <>
          <div style={{padding:'20px 48px 8px'}}>
            <span className="edo-cell-label" style={{color:'var(--edo-orange)'}}>03 · {lang==='fr'?'Type de packshot':'Packshot type'}</span>
          </div>
          <div style={{display:'grid',gridTemplateColumns:'repeat(3,1fr)',gap:1,background:'#000',borderTop:'1px solid #000',borderBottom:'1px solid #000'}}>
            {PAP_PACKSHOT_SUBS.map((sub,i)=>(
              <CfgChoice key={sub.k} idx={i+1} on={S.submethod===sub.k}
                onClick={()=>resetFrom('submethod', sub.k)}
                label={sub[lang]} desc={sub.desc[lang]}/>
            ))}
          </div>
        </>
      )}

      {/* Q2 — ACCESSOIRES sous-type */}
      {S.product === 'accessoires' && (
        <>
          <div style={{padding:'20px 48px 8px'}}>
            <span className="edo-cell-label" style={{color:'var(--edo-orange)'}}>02 · {lang==='fr'?"Type d'accessoire":'Accessory type'}</span>
          </div>
          <div style={{display:'grid',gridTemplateColumns:'repeat(3,1fr)',gap:1,background:'#000',borderTop:'1px solid #000',borderBottom:'1px solid #000'}}>
            {ACCESS_SUBS.map((sub,i)=>(
              <CfgChoice key={sub.k} idx={i+1} on={S.submethod===sub.k}
                onClick={()=>resetFrom('submethod', sub.k)}
                label={sub[lang]} desc={sub.desc[lang]}/>
            ))}
          </div>
        </>
      )}

      {/* Q3 — MÉDIA (on-model / accessoires / simple products) */}
      {((S.product === 'pap' && S.method === 'onmodel')
        || (S.product === 'accessoires' && S.submethod)
        || (['eyewear','food','cosmetique','bijoux'].includes(S.product))) && (
        <>
          <div style={{padding:'20px 48px 8px'}}>
            <span className="edo-cell-label" style={{color:'var(--edo-orange)'}}>
              {S.product==='pap' ? '03' : S.product==='accessoires' ? '03' : '02'} · {lang==='fr'?'Média':'Media'}
            </span>
          </div>
          <div style={{display:'grid',gridTemplateColumns:'1fr 1fr',gap:1,background:'#000',borderTop:'1px solid #000',borderBottom:'1px solid #000'}}>
            {MEDIA_OPTIONS.map((m,i)=>(
              <CfgChoice key={m.k} idx={i+1} on={S.media===m.k}
                onClick={()=>resetFrom('media', m.k)}
                label={m[lang]} desc={m.desc[lang]}/>
            ))}
          </div>
        </>
      )}

      {/* Q4 — PACKSHOT : vues (multi) */}
      {S.product === 'pap' && S.method === 'packshot' && S.submethod && (
        <>
          <div style={{padding:'20px 48px 8px'}}>
            <span className="edo-cell-label" style={{color:'var(--edo-orange)'}}>04 · {lang==='fr'?'Vues par produit':'Views per product'}</span>
            <span style={{fontFamily:'var(--font-mono)',fontSize:10,letterSpacing:'0.05em',color:'#888',marginLeft:12}}>
              {lang==='fr'?'(multi-sélection)':'(multi-select)'}
            </span>
          </div>
          <div style={{display:'grid',gridTemplateColumns:'repeat(auto-fit,minmax(140px,1fr))',gap:1,background:'#000',borderTop:'1px solid #000',borderBottom:'1px solid #000'}}>
            {PACKSHOT_VIEWS.filter(v => v.k !== '3/4' || S.submethod === 'ghost').map((v,i)=>{
              const on = (S.views || []).includes(v.k);
              return (
                <button key={v.k} onClick={()=>{
                  const cur = S.views || [];
                  setSession({ views: cur.includes(v.k) ? cur.filter(x => x !== v.k) : [...cur, v.k] });
                }} style={{
                  background:on?'#141414':'#fff',color:on?'#fff':'#141414',
                  border:0,padding:'16px',textAlign:'left',cursor:'pointer',fontFamily:'inherit',
                  display:'flex',flexDirection:'column',gap:6,minHeight:72,minWidth:0,
                }}>
                  <span style={{fontFamily:'var(--font-mono)',fontSize:10,letterSpacing:'0.15em',color:on?'rgba(255,255,255,0.6)':'#888'}}>
                    {String(i+1).padStart(2,'0')}
                  </span>
                  <span style={{fontSize:14,fontWeight:400,letterSpacing:'-0.015em'}}>{v[lang]}</span>
                  {on && <span style={{color:'var(--edo-orange)',fontSize:12,marginTop:'auto'}}>●</span>}
                </button>
              );
            })}
          </div>
          <div style={{background:'#fafafa',padding:'10px 48px',fontFamily:'var(--font-mono)',fontSize:10,letterSpacing:'0.05em',color:'#888',borderBottom:'1px solid #000'}}>
            {lang==='fr'?'"Détail" ajoute uniquement de la post-production — pas d\'impact sur la cadence.':'"Detail" only adds post-prod — no impact on throughput.'}
          </div>
        </>
      )}

      {/* Q5 — PACKSHOT : quantité */}
      {S.product === 'pap' && S.method === 'packshot' && S.submethod && (
        <div style={{display:'grid',gridTemplateColumns:'1fr',gap:1,background:'#000',borderTop:'1px solid #000',borderBottom:'1px solid #000'}}>
          <div style={{background:'#fff',padding:'18px 20px',display:'flex',flexDirection:'column',gap:12}}>
            <span className="edo-cell-label" style={{color:'#888'}}>{lang==='fr'?'Nombre de produits':'Number of products'}</span>
            <div style={{display:'flex',alignItems:'center',gap:10,maxWidth:320}}>
              <button onClick={()=>setSession({quantity: Math.max(0, (Number(S.quantity)||0)-1)})} style={stepperBtn}>−</button>
              <input value={S.quantity} onChange={e=>setSession({quantity: e.target.value.replace(/\D/g,'')})}
                placeholder="—" inputMode="numeric"
                style={{flex:1,background:'transparent',border:0,borderBottom:'1px solid var(--edo-gray-200)',outline:'none',padding:'4px 0',fontFamily:'var(--font-mono)',fontSize:24,letterSpacing:'-0.01em',color:'#141414',textAlign:'center'}}/>
              <button onClick={()=>setSession({quantity: (Number(S.quantity)||0)+1})} style={stepperBtn}>+</button>
            </div>
          </div>
        </div>
      )}

      {/* Q4+Q5 pour on-model / accessoires / simples : qty + viewsCount */}
      {((S.product === 'pap' && S.method === 'onmodel' && S.media)
        || (S.product === 'accessoires' && S.submethod && S.media)
        || (['eyewear','food','cosmetique','bijoux'].includes(S.product) && S.media)) && (
        <div style={{display:'grid',gridTemplateColumns:'1fr 1fr',gap:1,background:'#000',borderTop:'1px solid #000',borderBottom:'1px solid #000'}}>
          <div style={{background:'#fff',padding:'18px 20px',display:'flex',flexDirection:'column',gap:12}}>
            <span className="edo-cell-label" style={{color:'#888'}}>{lang==='fr'?'Nombre de produits':'Number of products'}</span>
            <div style={{display:'flex',alignItems:'center',gap:10}}>
              <button onClick={()=>setSession({quantity: Math.max(0, (Number(S.quantity)||0)-1)})} style={stepperBtn}>−</button>
              <input value={S.quantity} onChange={e=>setSession({quantity: e.target.value.replace(/\D/g,'')})}
                placeholder="—" inputMode="numeric"
                style={{flex:1,background:'transparent',border:0,borderBottom:'1px solid var(--edo-gray-200)',outline:'none',padding:'4px 0',fontFamily:'var(--font-mono)',fontSize:24,letterSpacing:'-0.01em',color:'#141414',textAlign:'center'}}/>
              <button onClick={()=>setSession({quantity: (Number(S.quantity)||0)+1})} style={stepperBtn}>+</button>
            </div>
          </div>
          <div style={{background:'#fff',padding:'18px 20px',display:'flex',flexDirection:'column',gap:12}}>
            <span className="edo-cell-label" style={{color:'#888'}}>{lang==='fr'?'Vues par produit':'Views per product'}</span>
            <div style={{display:'flex',alignItems:'center',gap:10}}>
              <button onClick={()=>setSession({viewsCount: Math.max(0, (Number(S.viewsCount)||0)-1)})} style={stepperBtn}>−</button>
              <input value={S.viewsCount} onChange={e=>setSession({viewsCount: e.target.value.replace(/\D/g,'')})}
                placeholder="—" inputMode="numeric"
                style={{flex:1,background:'transparent',border:0,borderBottom:'1px solid var(--edo-gray-200)',outline:'none',padding:'4px 0',fontFamily:'var(--font-mono)',fontSize:24,letterSpacing:'-0.01em',color:'#141414',textAlign:'center'}}/>
              <button onClick={()=>setSession({viewsCount: (Number(S.viewsCount)||0)+1})} style={stepperBtn}>+</button>
            </div>
          </div>
        </div>
      )}

      {/* POST-PROD */}
      {S.product && S.product !== 'cyclorama' && sessionValid(S) && (
        <>
          <div style={{padding:'20px 48px 8px'}}>
            <span className="edo-cell-label" style={{color:'var(--edo-orange)'}}>{lang==='fr'?'Post-production':'Post-production'}</span>
          </div>
          <div style={{display:'grid',gridTemplateColumns: (S.media === 'video' && S.postprod) ? '1fr 1fr' : '1fr',gap:1,background:'#000',borderTop:'1px solid #000',borderBottom:'1px solid #000'}}>
            <div style={{background:'#fff',padding:'16px 20px',display:'flex',alignItems:'center',justifyContent:'space-between',gap:12}}>
              <div>
                <div style={{fontSize:14,fontWeight:500,letterSpacing:'-0.01em'}}>{lang==='fr'?'Post-production par E-DO ?':'Post-production by E-DO?'}</div>
                <div style={{fontFamily:'var(--font-mono)',fontSize:10,color:'#888',marginTop:2}}>{lang==='fr'?'Chiffrage sur demande':'Quoted on request'}</div>
              </div>
              <CfgToggle on={S.postprod} onClick={()=>setSession({postprod:!S.postprod, postprodVideo: S.postprod ? false : S.postprodVideo})}/>
            </div>
            {(S.media === 'video') && S.postprod && (
              <div style={{background:'#fff',padding:'16px 20px',display:'flex',alignItems:'center',justifyContent:'space-between',gap:12}}>
                <div>
                  <div style={{fontSize:14,fontWeight:500,letterSpacing:'-0.01em'}}>{lang==='fr'?'Montage vidéo ?':'Video editing?'}</div>
                  <div style={{fontFamily:'var(--font-mono)',fontSize:10,color:'#888',marginTop:2}}>{lang==='fr'?'Uniquement pour les projets vidéo':'Only for video projects'}</div>
                </div>
                <CfgToggle on={S.postprodVideo} onClick={()=>setSession({postprodVideo:!S.postprodVideo})}/>
              </div>
            )}
          </div>
        </>
      )}

      {/* CYCLORAMA */}
      {S.product === 'cyclorama' && (
        <div style={{background:'#fafafa',padding:'40px 48px',borderTop:'1px solid #000',borderBottom:'1px solid #000',textAlign:'center'}}>
          <div style={{fontSize:17,fontWeight:400,letterSpacing:'-0.015em',marginBottom:8}}>
            {lang==='fr'?'Cyclorama / Production libre':'Cyclorama / Free production'}
          </div>
          <div style={{fontSize:13,color:'#595959',maxWidth:520,margin:'0 auto',lineHeight:1.5}}>
            {lang==='fr'
              ?"Besoin sur-mesure — nous établissons un devis personnalisé selon le plateau, la durée et les moyens techniques."
              :"Custom needs — we'll prepare a tailored quote based on stage, duration and technical resources."}
          </div>
        </div>
      )}

      {/* ADD ANOTHER (only visible after 1 valid session) */}
      {sessions.length === 1 && sessionValid(active) && (
        <div style={{padding:'24px 48px 0',display:'flex',justifyContent:'center'}}>
          <button onClick={addSession} style={{background:'#fff',border:'1px solid var(--edo-gray-200)',padding:'10px 18px',cursor:'pointer',fontFamily:'var(--font-mono)',fontSize:10,letterSpacing:'0.15em',textTransform:'uppercase',color:'#141414',display:'flex',alignItems:'center',gap:8,height:36}}>
            + {lang==='fr'?'Ajouter une autre session produit':'Add another product session'}
          </button>
        </div>
      )}

      {/* RECAP */}
      {allValid && recs && (
        <div style={{margin:'32px 0 0',background:'#141414',color:'#fff'}}>
          <div style={{padding:'24px 48px',borderBottom:'1px solid rgba(255,255,255,0.1)'}}>
            <span style={{fontFamily:'var(--font-mono)',fontSize:10,letterSpacing:'0.2em',textTransform:'uppercase',color:'var(--edo-orange)'}}>
              {lang==='fr'?'Recommandation':'Recommendation'}
            </span>
          </div>
          {recs.map((r, i) => {
            const px = BOOK_PLATEAUX.find(x => x.k === r.plateau);
            const p = PRODUCTS.find(x => x.k === r.session.product);
            const dur = r.onRequest ? (lang==='fr'?'sur demande':'on request')
              : r.slotType==='full' ? (r.totalDays > 1 ? `${r.totalDays} ${lang==='fr'?'journées':'days'}` : (lang==='fr'?'1 journée (8h)':'1 day (8h)'))
              : r.slotType==='half' ? `${lang==='fr'?'Demi-journée':'Half day'} (${r.hours}h)`
              : `${r.hours}h`;
            return (
              <div key={i} style={{padding:'20px 48px',borderBottom:i<recs.length-1 ? '1px solid rgba(255,255,255,0.1)':'none',display:'grid',gridTemplateColumns:'auto 1fr',gap:32,alignItems:'center'}}>
                <span style={{fontFamily:'var(--font-mono)',fontSize:11,letterSpacing:'0.2em',color:'rgba(255,255,255,0.5)'}}>
                  {String(i+1).padStart(2,'0')}
                </span>
                <div>
                  <div style={{fontSize:20,fontWeight:400,letterSpacing:'-0.02em',marginBottom:4}}>
                    {px[lang]} <span style={{color:'rgba(255,255,255,0.5)',fontSize:14}}>· {dur}</span>
                  </div>
                  <div style={{fontFamily:'var(--font-mono)',fontSize:10,letterSpacing:'0.05em',color:'rgba(255,255,255,0.55)'}}>
                    {p?.[lang]} · {r.session.quantity} {lang==='fr'?'produits':'products'}
                    {r.cadence ? ` · ${r.cadence}/${lang==='fr'?'jour':'day'}` : ''}
                  </div>
                </div>
              </div>
            );
          })}
          <div style={{padding:'28px 48px',display:'flex',alignItems:'center',justifyContent:'space-between',gap:16,flexWrap:'wrap'}}>
            <div style={{fontSize:12,color:'rgba(255,255,255,0.55)',maxWidth:560,lineHeight:1.5}}>
              {lang==='fr'
                ?"Vous pourrez ajuster chaque plateau avant validation finale."
                :"You'll be able to tweak every stage before final booking."}
            </div>
            <button onClick={onApply} style={{
              background:'var(--edo-orange)',border:0,color:'#fff',padding:'14px 28px',cursor:'pointer',
              fontFamily:'var(--font-mono)',fontSize:11,letterSpacing:'0.2em',textTransform:'uppercase',
              display:'inline-flex',alignItems:'center',gap:10,
            }}>
              {lang==='fr'?'Continuer vers la réservation':'Continue to booking'}
              <IconArrowRight width="14" height="14"/>
            </button>
          </div>
        </div>
      )}

      <div style={{height:32,background:'#fff'}}/>
    </div>
  );
};
